Introduction

Looking at the conditions of refugee camps across 36 countries and 227 locations. The UNHCR Information Management System (IRIS) collected the data as an aggregate of conditions across specfic camps. I want to use this data to visualize these conditions and make cross-comparisons of similarities and differences based on location, time, and reason.

Clean the Data

# drop rows that don't have substantial data
wash_data_lim <- unhcr_wash_data %>% 
  mutate(toilets_per_HH_percentage = coalesce(households_with_toilet, household_toilet)) %>% 
  mutate(region = countrycode(country,
         origin = "country.name",
         destination = "un.regionsub.name")) |> 
  mutate(region = factor(region, levels = c("Latin America and the Caribbean",
                                            "Northern Africa",
                                            "Sub-Saharan Africa",
                                            "Western Asia",
                                            "Southern Asia"))) |> 
  filter(!if_all(persons_per_handpump:reporting_annual, is.na)) %>% 
  select(-c(form_id, 
            location_id, 
            reporting_monthly, 
            reporting_annual, 
            household_toilet, 
            households_with_toilet)) %>% 
  relocate(refugee_pop, .before = persons_per_handpump) %>% 
  relocate(c(persons_per_shower, 
             persons_per_hygiene_promoter, 
             persons_per_toilet), .before = liters_per_person_per_day) %>% 
  relocate(liters_per_person_household, .after = liters_per_person_per_day) %>% 
  relocate(c(defecate_in_toilet, 
             toilets_per_HH_percentage, 
             solid_waste_disposal_access, 
             menstrual_hygiene_satisfaction, 
             access_to_soap), .after = persons_per_toilet) %>% 
  rename(safe_water_perc = chlorinated_safe_water_quality,
         liters_per_HH_per_day = liters_per_person_household,
         storage10L_HH_perc = potable_water_storage_10l,
         safe_water_collect_HH_perc = protected_water_sources,
         defecate_in_toilet_HH_perc = defecate_in_toilet,
         access_to_soap_HH_perc = access_to_soap,
         solid_waste_disposal_HH_perc = solid_waste_disposal_access) %>% 
  relocate(safe_water_perc, .before = safe_water_collect_HH_perc) |> 
  relocate(c(region, country), .before = location_name)

# get a new year column so for better summary
wash_data_lim <- wash_data_lim |> 
  mutate(year = year(start_date),
         month = month(start_date + (end_date - start_date) / 2))

wash_data_lim <- wash_data_lim |> 
  mutate(report_date = sprintf("%d-%02d", year, month)) |> 
  relocate(report_date, .before = start_date)
# some are a bit weird, but when checking against later dataset, they are excluded

wash_data_lim <- wash_data_lim |> 
  select(-c(year, month))

  filter(not_NA > 6) |> 
  relocate(c(liters_per_person_per_day, safe_water_perc), .before = persons_per_shower) |>
